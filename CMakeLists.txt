cmake_minimum_required(VERSION 3.20)
project(GakuFSR4 LANGUAGES CXX)
enable_testing()
add_library(gaku_imgui STATIC
  vendor/imgui/imgui.cpp vendor/imgui/imgui_draw.cpp
  vendor/imgui/imgui_tables.cpp vendor/imgui/imgui_widgets.cpp)
target_include_directories(gaku_imgui PUBLIC vendor/imgui)
set_property(TARGET gaku_imgui PROPERTY MSVC_RUNTIME_LIBRARY MultiThreaded)
add_library(GakuFSR4 SHARED src/GakuFSR4.cpp vendor/ghpc/present.cpp src/SettingsUi.cpp src/NativeSwapchain.cpp)
target_compile_features(GakuFSR4 PRIVATE cxx_std_17)
target_include_directories(GakuFSR4 PRIVATE include
  vendor/ghpc/ffx/Kits/FidelityFX/api/include
  vendor/ghpc/ffx/Kits/FidelityFX/upscalers/include
  vendor/ghpc/ffx/Kits/FidelityFX/framegeneration/include)
target_compile_definitions(GakuFSR4 PRIVATE WIN32_LEAN_AND_MEAN UNICODE _UNICODE PUBLIC=1)
target_compile_options(GakuFSR4 PRIVATE /W4 /EHa /utf-8)
target_link_libraries(GakuFSR4 PRIVATE d3d11 d3d12 dxgi gaku_imgui)
set_property(TARGET GakuFSR4 PROPERTY MSVC_RUNTIME_LIBRARY MultiThreaded)
add_executable(test_math src/test_math.cpp)
target_compile_features(test_math PRIVATE cxx_std_17)
target_compile_options(test_math PRIVATE /UNDEBUG)
add_test(NAME frame_math COMMAND test_math)
add_executable(test_native_aa_math tests/native_aa_unit.cpp)
target_compile_features(test_native_aa_math PRIVATE cxx_std_17)
target_include_directories(test_native_aa_math PRIVATE src)
target_compile_options(test_native_aa_math PRIVATE /UNDEBUG)
add_test(NAME native_aa_math COMMAND test_native_aa_math)
add_executable(test_dafp src/test_dafp.cpp)
target_compile_features(test_dafp PRIVATE cxx_std_17)
target_compile_options(test_dafp PRIVATE /UNDEBUG)
add_test(NAME dafp_diagnostics COMMAND test_dafp)
add_executable(test_fg_distortion src/test_fg_distortion.cpp)
target_compile_features(test_fg_distortion PRIVATE cxx_std_17)
target_compile_options(test_fg_distortion PRIVATE /UNDEBUG)
add_test(NAME fg_distortion_math COMMAND test_fg_distortion)
add_executable(test_gpu src/test_gpu.cpp src/TestBackbufferFormat.cpp)
target_compile_features(test_gpu PRIVATE cxx_std_17)
target_link_libraries(test_gpu PRIVATE d3d11 dxgi d3dcompiler)
add_executable(test_native_swapchain src/test_native_swapchain.cpp)
target_compile_features(test_native_swapchain PRIVATE cxx_std_17)
target_include_directories(test_native_swapchain PRIVATE
  vendor/ghpc/ffx/Kits/FidelityFX/api/include
  vendor/ghpc/ffx/Kits/FidelityFX/framegeneration/include)
target_link_libraries(test_native_swapchain PRIVATE d3d11 d3d12 dxgi)
add_executable(test_plugin src/test_plugin.cpp)
target_include_directories(test_plugin PRIVATE include)
target_link_libraries(test_plugin PRIVATE gaku_imgui)
set_property(TARGET test_plugin PROPERTY MSVC_RUNTIME_LIBRARY MultiThreaded)
add_test(NAME plugin_abi_and_ui COMMAND test_plugin "$<TARGET_FILE:GakuFSR4>" "${CMAKE_CURRENT_BINARY_DIR}")
add_executable(test_plugin_event src/test_plugin_event.cpp)
target_compile_features(test_plugin_event PRIVATE cxx_std_17)
add_test(NAME plugin_event_binding COMMAND test_plugin_event)
add_executable(test_managed_root src/test_managed_root.cpp)
target_compile_features(test_managed_root PRIVATE cxx_std_17)
add_test(NAME managed_root COMMAND test_managed_root)
